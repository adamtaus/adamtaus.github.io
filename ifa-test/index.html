#include #include #include #include #include #include // LCD I2C #define LCD_SDA 21 #define LCD_SCL 22 // HX711 #define HX_DT 4 #define HX_SCK 16 // Servo #define SERVO_PIN 17 // Bzucak #define BUZZER_PIN 25 // RFID RC522 #define RFID_SS_PIN 5 #define RFID_RST_PIN 27 #define RFID_SCK 18 #define RFID_MISO 19 #define RFID_MOSI 23 LiquidCrystal_I2C lcd(0x27, 16, 2); HX711 scale; Servo gateServo; MFRC522 rfid(RFID_SS_PIN, RFID_RST_PIN); // Váha float calibration_factor = -7050.0; // TEST cíl int targetWeight = 0; const int TEST_TARGET_WEIGHT = 5; int currentWeight = 0; const int tolerance = 4; // Servo pozice const int SERVO_BLOCK = 10; const int SERVO_RELEASE = 95; // Čas držení const unsigned long HOLD_TIME = 20000UL; bool solved = false; bool matchRunning = false; bool waitingForClearScale = false; unsigned long matchStart = 0; int lastCountdownSecond = -1; // Aktualizace LCD unsigned long lastDisplayUpdate = 0; const unsigned long DISPLAY_INTERVAL = 400; // Kontrola prázdné váhy unsigned long emptySince = 0; const unsigned long EMPTY_CONFIRM_TIME = 2000; // Tvoje servisní karta const byte AUTHORIZED_UID_COUNT = 4; byte authorizedUIDs[AUTHORIZED_UID_COUNT][4] = { {0x37, 0x16, 0x72, 0x03}, {0x95, 0x00, 0x72, 0x03}, {0xC7, 0xAF, 0xCF, 0x06}, {0xCD, 0xA2, 0x12, 0x06} }; void setup() { Serial.begin(115200); delay(500); Wire.begin(LCD_SDA, LCD_SCL); SPI.begin(RFID_SCK, RFID_MISO, RFID_MOSI, RFID_SS_PIN); lcd.init(); lcd.backlight(); pinMode(BUZZER_PIN, OUTPUT); noTone(BUZZER_PIN); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Startuji..."); scale.begin(HX_DT, HX_SCK); scale.set_scale(calibration_factor); scale.tare(); rfid.PCD_Init(); rfid.PCD_AntennaOn(); closeGate(); resetRound(); playStartupSound(); delay(1000); } void loop() { checkRFID(); float reading = scale.get_units(5); if (reading < 0) reading = 0; currentWeight = (int)round(reading); if (waitingForClearScale) { if (millis() - lastDisplayUpdate >= DISPLAY_INTERVAL) { lastDisplayUpdate = millis(); printLine(0, "Splneno!"); printLine(1, "Sundej predmet"); } if (currentWeight <= 3) { if (emptySince == 0) { emptySince = millis(); } if (millis() - emptySince >= EMPTY_CONFIRM_TIME) { scale.tare(); resetRound(); } } else { emptySince = 0; } delay(100); return; } bool weightMatches = abs(currentWeight - targetWeight) <= tolerance && !solved; if (weightMatches) { if (!matchRunning) { matchRunning = true; matchStart = millis(); lastCountdownSecond = -1; lastDisplayUpdate = 0; playMatchStartSound(); } unsigned long heldTime = millis() - matchStart; unsigned long remainingTime = heldTime >= HOLD_TIME ? 0 : HOLD_TIME - heldTime; int secondsLeft = (remainingTime + 999) / 1000; if (heldTime >= HOLD_TIME) { solved = true; releaseBall(); playVictoryMelody(); waitingForClearScale = true; matchRunning = false; lastCountdownSecond = -1; emptySince = 0; lastDisplayUpdate = 0; delay(120); return; } if (secondsLeft != lastCountdownSecond) { playCountdownTick(secondsLeft); lastCountdownSecond = secondsLeft; } if (millis() - lastDisplayUpdate >= DISPLAY_INTERVAL) { lastDisplayUpdate = millis(); printLine(0, "Spravna hmotnost!"); printLine(1, String("Drz: ") + secondsLeft + " s"); } } else { matchRunning = false; lastCountdownSecond = -1; int remaining = targetWeight - currentWeight; if (millis() - lastDisplayUpdate >= DISPLAY_INTERVAL) { lastDisplayUpdate = millis(); printLine(0, String("Cil: ") + targetWeight + " g"); printLine(1, String("Zbyva: ") + remaining + " g"); } } delay(120); } void printLine(byte row, String text) { if (text.length() > 16) { text = text.substring(0, 16); } while (text.length() < 16) { text += " "; } lcd.setCursor(0, row); lcd.print(text); } void checkRFID() { if (!rfid.PICC_IsNewCardPresent()) return; if (!rfid.PICC_ReadCardSerial()) return; Serial.print("UID: "); for (byte i = 0; i < rfid.uid.size; i++) { Serial.print("0x"); if (rfid.uid.uidByte[i] < 0x10) Serial.print("0"); Serial.print(rfid.uid.uidByte[i], HEX); Serial.print(" "); } Serial.println(); if (isAuthorizedCard()) { playAccessGrantedSound(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("SERVIS RFID"); lcd.setCursor(0, 1); lcd.print("Poustim..."); delay(600); releaseBall(); playVictoryMelody(); waitingForClearScale = true; matchRunning = false; lastCountdownSecond = -1; emptySince = 0; lastDisplayUpdate = 0; } else { playAccessDeniedSound(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("RFID nacteno"); lcd.setCursor(0, 1); lcd.print("Neplatny cip"); delay(1000); lcd.clear(); } rfid.PICC_HaltA(); rfid.PCD_StopCrypto1(); } bool isAuthorizedCard() { if (rfid.uid.size != 4) return false; for (byte card = 0; card < AUTHORIZED_UID_COUNT; card++) { bool match = true; for (byte i = 0; i < 4; i++) { if (rfid.uid.uidByte[i] != authorizedUIDs[card][i]) { match = false; break; } } if (match) { return true; } } return false; } void playNote(int frequency, int durationMs) { if (frequency > 0) { tone(BUZZER_PIN, frequency); } delay(durationMs); noTone(BUZZER_PIN); delay(25); } void playStartupSound() { playNote(784, 70); playNote(988, 70); playNote(1175, 120); } void playMatchStartSound() { playNote(880, 45); playNote(1320, 70); } void playCountdownTick(int secondsLeft) { if (secondsLeft <= 0 || secondsLeft == 20) return; if (secondsLeft <= 5) { playNote(1200 + (5 - secondsLeft) * 90, 35); } else if (secondsLeft % 5 == 0) { playNote(1047, 45); } } void playAccessGrantedSound() { playNote(988, 60); playNote(1319, 90); } void playAccessDeniedSound() { playNote(220, 120); playNote(0, 40); playNote(196, 160); } void playVictoryMelody() { const int melody[] = {523, 659, 784, 1047, 784, 1047, 1319}; const int durations[] = {90, 90, 90, 160, 90, 120, 260}; const byte noteCount = sizeof(melody) / sizeof(melody[0]); for (byte i = 0; i < noteCount; i++) { playNote(melody[i], durations[i]); } } void closeGate() { gateServo.setPeriodHertz(50); gateServo.attach(SERVO_PIN, 500, 2400); delay(50); gateServo.write(SERVO_BLOCK); delay(700); gateServo.detach(); } void releaseBall() { lcd.clear(); lcd.setCursor(0, 0); lcd.print("Poustim"); lcd.setCursor(0, 1); lcd.print("kulicku..."); gateServo.setPeriodHertz(50); gateServo.attach(SERVO_PIN, 500, 2400); delay(50); gateServo.write(SERVO_RELEASE); delay(2500); gateServo.write(SERVO_BLOCK); delay(700); gateServo.detach(); } void resetRound() { targetWeight = TEST_TARGET_WEIGHT; solved = false; matchRunning = false; waitingForClearScale = false; matchStart = 0; lastCountdownSecond = -1; emptySince = 0; lastDisplayUpdate = 0; lcd.clear(); lcd.setCursor(0, 0); lcd.print("Novy cil"); lcd.setCursor(0, 1); lcd.print("Cil: "); lcd.print(targetWeight); lcd.print(" g"); delay(1000); lcd.clear(); }
